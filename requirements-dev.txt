# Development tooling. Deliberately separate from requirements.txt, which the
# Dockerfile installs — neither a linter nor a coverage tool belongs in the
# production image.
#
#   pip install -r requirements-dev.txt
#   ruff format .         # format in place
#   ruff check .          # lint — after formatting, which can move a '# noqa'
#
# Pinned to the exact version the Ruff CI job runs (.github/workflows/ruff.yml),
# so a new ruff release cannot turn an unrelated pull request red. Bumping it is
# a deliberate commit, not a surprise.
ruff==0.16.2

# Coverage runs locally and in CI — .github/workflows/test.yml installs this file
# for it, gates the build on `coverage report` and writes the badge from the same
# data (#272). So this pin keeps successive measurements comparable *and* keeps a
# workflow green: a coverage release that changed how branches are counted would
# move the number the gate judges.
#
#   coverage run manage.py test projects --noinput
#   coverage report
#   coverage html   # browsable, writes htmlcov/index.html
#
# No flags needed: what is measured, what is excluded and where the floor sits
# all live in the [tool.coverage.*] sections of pyproject.toml, so a plain report
# is already the honest number and the same check CI runs. Both artefacts
# (.coverage, htmlcov/) are gitignored.
coverage==7.16.0
